# Три внешние зависимости; причины выбора зафиксированы здесь рядом с каждой.
# Всё остальное (разбор контейнера 1С, справки, выгрузок, граф) — стандартная
# библиотека.
#
# Официальный SDK обеспечивает протокол и совместимый transport. Самописный
# аналог уже приводил к подключению через обёртку curl вокруг HTTP-эндпоинта.
mcp>=2.0

# Постинги поискового индекса лежат массивами, а не словарём словарей: пара
# «документ, вес» стоит 12 байт вместо ~100, и на трёх конфигурациях со
# справкой это 183 -> 45 МБ. Заодно начисление стало векторным: поиск по
# справке 2,3 -> 1,1 мс. Стандартной библиотекой не закрывается — `array` даёт
# ту же память, но цикл начисления пришлось бы писать на Python, и поиск стал
# бы медленнее. Нужен `bitwise_count` из numpy 2.0.
numpy>=2.0

# Стемминг русского. Было — обрезка токена до шести символов: она сводит
# «контрагент/контрагенты», но не «чек/чека/чеков», «вид/виды», «оплат/оплаты»,
# и запрос «виды оплат чеков» не находил `ВидыОплатЧекаККМ` вовсе. На восьми
# запросах со склонениями первым местом: 1/8 -> 4/8. На автоматических наборах
# (11 665 запросов по метаданным, 50 572 по справке) — ни одного изменения,
# то есть регресса нет. Заодно быстрее: поиск по метаданным 0,097 -> 0,085 мс,
# по справке 0,281 -> 0,266 — основ меньше (1844 -> 1751), постинги короче.
# Морфологии русского в стандартной библиотеке нет. 102 КБ, чистый Python, без
# транзитивных зависимостей и без компиляции; C-версия `PyStemmer` (242 КБ)
# на наших данных не быстрее, `pymorphy3` со словарём — 8,2 МБ и на нашем
# словаре хуже (7 групп из 8 против 8 из 8).
snowballstemmer>=3.0
